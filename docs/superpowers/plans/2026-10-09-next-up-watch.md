# Next Up for Apple Watch Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show the countdown to the next campground check-in as Apple Watch complications (circular, corner, rectangular, inline) and in a small read-only watch app, fed by the iPhone app over WatchConnectivity.

**Architecture:** A single-target watchOS app (`NextUpWatch`) with a WidgetKit extension (`NextUpWatchWidgetExtension`), embedded in the iPhone app. Both compile `Shared/`; the extension also compiles `WidgetShared/`, so the existing `Provider` and accessory views do the work. The iPhone app pushes a `WatchPayload` (feeds plus latest snapshot) after every refresh; the watch app stores it in its own App Group container, where the complication's `Provider` reads it and fetches on its own when stale.

**Tech Stack:** Swift 5 language mode, SwiftUI, WidgetKit, WatchConnectivity, Swift Testing, XcodeGen 2.46.0, Fastlane (match, produce, pilot), GitHub Actions on `macos-15`.

**Spec:** `docs/superpowers/specs/2026-10-09-next-up-watch-design.md` (builds on `docs/superpowers/specs/2026-10-09-next-up-ios-design.md`).

```swift
import Foundation

/// What the iPhone app sends the watch, which can't see the phone's App Group:
/// the feeds (the phone owns them) and its latest fetch.
struct WatchPayload: Codable, Sendable {
    static let key = "payload"

    var feeds: [CalendarFeed]
    var snapshot: StaySnapshot?

    /// WatchConnectivity dictionaries only carry property-list values, so the payload travels as JSON.
    func dictionary() -> [String: Any] {
        guard let data = try? JSONEncoder().encode(self) else { return [:] }
        return [Self.key: data]
    }

    /// The watch fetches on its own too, so a push can arrive older than what it already has.
    static func newer(_ incoming: StaySnapshot?, than current: StaySnapshot?) -> StaySnapshot? {
        guard let incoming else { return current }
        guard let current else { return incoming }
        return incoming.fetchedAt > current.fetchedAt ? incoming : current
    }
}

extension WatchPayload {
    init?(dictionary: [String: Any]) {
        guard let data = dictionary[Self.key] as? Data,
              let decoded = try? JSONDecoder().decode(Self.self, from: data)
        else { return nil }
        self = decoded
    }
}
```

```swift
import Foundation
import WatchConnectivity

/// Keeps the watch's copy of the feeds and stays current. The watch has its own App Group container,
/// so it can't read the phone's.
final class PhoneWatchSync: NSObject, WCSessionDelegate, @unchecked Sendable {
    static let shared = PhoneWatchSync()

    func activate() {
        guard WCSession.isSupported() else { return }
        WCSession.default.delegate = self
        WCSession.default.activate()
    }

    func send(feeds: [CalendarFeed], snapshot: StaySnapshot?) {
        guard WCSession.isSupported() else { return }
        let session = WCSession.default
        guard session.activationState == .activated, session.isPaired, session.isWatchAppInstalled else { return }
        let payload = WatchPayload(feeds: feeds, snapshot: snapshot).dictionary()
        // A failed update gets retried by the next refresh.
        try? session.updateApplicationContext(payload)
        // Context waits for the watch app's next launch; this wakes it now, within a daily budget.
        if session.isComplicationEnabled, session.remainingComplicationUserInfoTransfers > 0 {
            session.transferCurrentComplicationUserInfo(payload)
        }
    }

    private func sendStored() {
        send(feeds: SharedStore.feeds, snapshot: SharedStore.snapshot)
    }

    func session(_ session: WCSession, activationDidCompleteWith activationState: WCSessionActivationState, error: Error?) {
        // A freshly installed watch app shouldn't wait for the next refresh.
        sendStored()
    }

    func sessionWatchStateDidChange(_ session: WCSession) {
        sendStored()
    }

    func sessionDidBecomeInactive(_ session: WCSession) {}

    func sessionDidDeactivate(_ session: WCSession) {
        // The user switched watches; the new one needs a session too.
        session.activate()
    }
}
```

```swift
import Foundation
import WatchConnectivity
import WidgetKit

/// Stores what the iPhone sends where the complication's provider reads it.
final class WatchReceiver: NSObject, WCSessionDelegate, @unchecked Sendable {
    static let shared = WatchReceiver()

    /// Lets the open app redraw when new data lands.
    @MainActor var onUpdate: (@MainActor () -> Void)?

    func activate() {
        WCSession.default.delegate = self
        WCSession.default.activate()
    }

    func session(_ session: WCSession, activationDidCompleteWith activationState: WCSessionActivationState, error: Error?) {
        // Picks up context that arrived while the app wasn't running.
        if activationState == .activated { apply(session.receivedApplicationContext) }
    }

    func session(_ session: WCSession, didReceiveApplicationContext applicationContext: [String: Any]) {
        apply(applicationContext)
    }

    func session(_ session: WCSession, didReceiveUserInfo userInfo: [String: Any] = [:]) {
        apply(userInfo)
    }

    private func apply(_ dictionary: [String: Any]) {
        guard let payload = WatchPayload(dictionary: dictionary) else { return }
        SharedStore.feeds = payload.feeds
        SharedStore.snapshot = WatchPayload.newer(payload.snapshot, than: SharedStore.snapshot)
        WidgetCenter.shared.reloadAllTimelines()
        Task { @MainActor in self.onUpdate?() }
    }
}
```

- [ ] **Step 2: `NextUpWatch/WatchAppModel.swift`:**

```swift
import Foundation
import Observation

@MainActor
@Observable
final class WatchAppModel {
    private(set) var snapshot: StaySnapshot? = SharedStore.snapshot
    private(set) var hasFeeds = !SharedStore.feeds.isEmpty
    private(set) var isRefreshing = false
    private(set) var now = Date.now

    var upcomingStays: [Stay] { snapshot?.stays.upcoming(from: now) ?? [] }

    func reload() {
        snapshot = SharedStore.snapshot
        hasFeeds = !SharedStore.feeds.isEmpty
        now = .now
    }

    /// The phone usually keeps the watch current; this is for when it's out of reach.
    func refresh() async {
        let feeds = SharedStore.feeds
        guard !isRefreshing, feeds.contains(where: \.isEnabled) else { return }
        isRefreshing = true
        defer { isRefreshing = false }
        snapshot = await FeedRefresh.run(feeds: feeds, previous: SharedStore.snapshot)
        now = .now
    }
}
```

```swift
import SwiftUI

struct WatchContentView: View {
    @Environment(WatchAppModel.self) private var model

    var body: some View {
        NavigationStack {
            List {
                if model.upcomingStays.isEmpty {
                    Text(model.hasFeeds ? "No upcoming stays." : "Add a feed in Next Up on your iPhone.")
                        .foregroundStyle(.secondary)
                }
                ForEach(model.upcomingStays) { stay in
                    VStack(alignment: .leading, spacing: 2) {
                        Text(StayFormatting.countdown(stay, now: model.now))
                            .font(.caption.weight(.semibold))
                            .foregroundStyle(.tint)
                        Text(stay.title)
                            .font(.headline)
                            .lineLimit(2)
                        if let place = stay.place {
                            Text(place)
                                .font(.caption2)
                                .foregroundStyle(.secondary)
                                .lineLimit(1)
                        }
                        Text("\(StayFormatting.shortDates(stay)) · \(StayFormatting.nights(stay))")
                            .font(.caption2)
                            .foregroundStyle(.secondary)
                    }
                }
                if let fetchedAt = model.snapshot?.fetchedAt {
                    Text("Updated \(fetchedAt, format: .relative(presentation: .named))")
                        .font(.footnote)
                        .foregroundStyle(.secondary)
                        .listRowBackground(Color.clear)
                }
            }
            .navigationTitle("Next Up")
            .refreshable { await model.refresh() }
        }
    }
}
```

```swift
import SwiftUI

@main
struct NextUpWatchApp: App {
    @State private var model = WatchAppModel()
    @Environment(\.scenePhase) private var scenePhase

    init() {
        // Before any UI: a background launch from a complication transfer delivers only once a session is active.
        WatchReceiver.shared.activate()
    }

    var body: some Scene {
        WindowGroup {
            WatchContentView()
                .environment(model)
                .onAppear { WatchReceiver.shared.onUpdate = { [model] in model.reload() } }
        }
        .onChange(of: scenePhase, initial: true) { _, phase in
            if phase == .active { model.reload() }
        }
    }
}
```
