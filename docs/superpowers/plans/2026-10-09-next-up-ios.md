`Shared/FeedEditing.swift`:
```swift
import Foundation

extension CalendarFeed {
    /// Accepts http(s) and webcal(s) links with a host; anything else is a typo worth catching before saving.
    static func validatedURL(_ text: String) -> URL? {
        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard let url = URL(string: trimmed),
              let scheme = url.scheme?.lowercased(),
              ["http", "https", "webcal", "webcals"].contains(scheme),
              url.host() != nil
        else { return nil }
        return url
    }

    /// A new feed, or `existing` with a new name and URL. Nil when the URL isn't valid.
    static func edited(name: String, urlText: String, from existing: CalendarFeed?) -> CalendarFeed? {
        guard let url = validatedURL(urlText) else { return nil }
        let trimmedName = name.trimmingCharacters(in: .whitespacesAndNewlines)
        let displayName = trimmedName.isEmpty ? (url.host() ?? "Calendar") : trimmedName
        guard var feed = existing else { return CalendarFeed(name: displayName, url: url) }
        feed.name = displayName
        feed.url = url
        return feed
    }
}

extension Array where Element == CalendarFeed {
    func upserting(_ feed: CalendarFeed) -> [CalendarFeed] {
        guard let index = firstIndex(where: { $0.id == feed.id }) else { return self + [feed] }
        var copy = self
        copy[index] = feed
        return copy
    }

    func removing(_ ids: Set<UUID>) -> [CalendarFeed] {
        filter { !ids.contains($0.id) }
    }
}
```

import Foundation

enum RefreshRule {
    /// WidgetKit throttles reloads below about 15 minutes anyway.
    static let minimumMinutes = 15

    static func widgetShouldFetch(snapshotAge: TimeInterval?, refreshMinutes: Int, hasEnabledFeeds: Bool) -> Bool {
        guard hasEnabledFeeds else { return false }
        guard let snapshotAge else { return true }
        return snapshotAge >= TimeInterval(Swift.max(minimumMinutes, refreshMinutes) * 60)
    }
}

    /// Lock Screen countdown: "Today", "in 3d". Never "Tomorrow"; it doesn't fit beside a date.
    private static func compactCountdown(_ stay: Stay, now: Date, calendar: Calendar) -> String {

import BackgroundTasks
import Foundation
import Observation

@MainActor
@Observable
final class IOSAppModel {
    nonisolated static let refreshTaskID = (Bundle.main.bundleIdentifier ?? "biz.fitz.NextUp") + ".refresh"

    private(set) var feeds: [CalendarFeed] = SharedStore.feeds
    private(set) var snapshot: StaySnapshot? = SharedStore.snapshot
    private(set) var isRefreshing = false
    private(set) var hasLoaded = false
    private(set) var now = Date.now

    var upcomingStays: [Stay] { snapshot?.stays.upcoming(from: now) ?? [] }

    func status(for feed: CalendarFeed) -> FeedStatus? { snapshot?.statuses[feed.id] }

    func save(_ feed: CalendarFeed) { setFeeds(feeds.upserting(feed)) }
    func remove(_ ids: Set<UUID>) { setFeeds(feeds.removing(ids)) }
    func setEnabled(_ feed: CalendarFeed, _ enabled: Bool) {
        var updated = feed
        updated.isEnabled = enabled
        save(updated)
    }

    private func setFeeds(_ newFeeds: [CalendarFeed]) {
        feeds = newFeeds
        SharedStore.feeds = newFeeds
        Task { await refresh() }
    }

    /// The widget fetches on its own; opening the app is a cheap extra so the list is current.
    func becameActive() async {
        now = .now
        feeds = SharedStore.feeds
        snapshot = SharedStore.snapshot
        hasLoaded = true
        await refresh()
    }

    func refresh() async {
        guard !isRefreshing, feeds.contains(where: \.isEnabled) else { return }
        isRefreshing = true
        defer { isRefreshing = false }
        snapshot = await FeedRefresh.run(feeds: feeds, previous: snapshot)
        now = .now
    }

    /// iOS decides when, or whether, this runs; it only ever helps the widget's own fetching.
    func scheduleBackgroundRefresh() {
        let request = BGAppRefreshTaskRequest(identifier: Self.refreshTaskID)
        request.earliestBeginDate = .now + TimeInterval(max(RefreshRule.minimumMinutes, SharedStore.settings.refreshInterval) * 60)
        try? BGTaskScheduler.shared.submit(request)
    }

    func backgroundRefresh() async {
        feeds = SharedStore.feeds
        await refresh()
        scheduleBackgroundRefresh()
    }
}
